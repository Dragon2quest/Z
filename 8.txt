#include <stdint.h>

uint32_t calculateFactorial(uint32_t n) {
    uint32_t result = 1;
    uint32_t i;
    
    for (i = 1; i <= n; i++) {
        result = result * i;
    }
    
    return result;
}

int main(void) {
    volatile uint32_t targetNumber = 5;
    volatile uint32_t finalFactorial = 0;

    finalFactorial = calculateFactorial(targetNumber);

    while (1) {
    }
}
